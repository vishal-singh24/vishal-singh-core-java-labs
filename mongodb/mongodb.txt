1. Display all documents in the restaurants collection.

   db["restaurants"].find()

2. Display only restaurant_id, name, borough, and cuisine for all restaurants.

   db["restaurants"].find({},{restaurant_id:1,name:1,borough:1,cuisine:1})

3. Same as above, but exclude _id from the results.

   db["restaurants"].find({},{restaurant_id:1,name:1,borough:1,cuisine:1,_id:0})

4. Display restaurant_id, name, borough, and zipcode — again without _id.

   db["restaurants"].find({},{restaurant_id:1,name:1,borough:1,address:{zipcode:1},_id:0})

5. Find all restaurants located in the Bronx.

   db["restaurants"].find({borough:"Bronx"})

6. Show only the first 5 restaurants in the Bronx.

   db.restaurants.find({borough:"Bronx"}).limit(5)
  

7. Skip the first 5 Bronx restaurants and show the next 5.

   db.restaurants.find({borough:"Bronx"}).skip(5).limit(5)

8.  Find restaurants that achieved a score greater than 90.

    db.restaurants.find({"grades.score":{$gt:90}})

9. Find restaurants with a score between 80 and 100 (exclusive).

    db.restaurants.find({"grades.score":{$gt:80,$lt:100}})
    db.restaurants.find({grades:{$elemMatch:{score:{$gt:80,$lt:100}}}})

10. Find restaurants with a latitude value less than -95.754168.

    db.restaurants.find({"address.coord.1":{$lt:-95.754168}})

11. Find restaurants that:

	Do not serve American cuisine
	Have a score greater than 70
	Are located at a latitude less than -65.754168

    db.restaurants.find({$and:[{cuisine:{$ne:"American"}},{"grades.score":{$gt:70}},{"address.coord.0":{$lt:-65.754168}}]})
    

12. Same as Q11, but without using the $and operator.

     db.restaurants.find({cuisine:{$ne:"American"},"grades.score":{$gt:70},"address.coord":{$lt:-65.754168}})

Q13. Find restaurants that:

     Do not serve American cuisine
     Have a grade of 'A'
     Are not in Brooklyn
     Sort results by cuisine in descending order.

Q14. Find restaurant_id, name, borough, and cuisine for restaurants whose name starts with 'Wil'.

     db.restaurants.find({ name: { $regex: /^Will/ } },{restaurant_id:1,name:1,borough:1,cuisine:1})

Q15. Same fields for restaurants whose name ends with 'ces'.

     db.restaurants.find({ name: { $regex: /ces$/ } },{restaurant_id:1,name:1,borough:1,cuisine:1})

Q16. Same fields for restaurants whose name contains 'Reg' anywhere.

     db.restaurants.find({ name: { $regex: /Reg/ } },{restaurant_id:1,name:1,borough:1,cuisine:1})

17. Find restaurants in the Bronx that serve either American or Chinese cuisine.

     db.restaurants.find({borough:"Bronx",cuisine:{$in:["American","Chinese"]}})

18. Display restaurant_id, name, borough, and cuisine for restaurants in Staten Island, Queens, Bronx, or Brooklyn.

     db.restaurants.find({borough:{$in:["Staten Island","Queens","Bronx","Brooklyn"]}},{restaurant_id:1,name:1,borough:1,cuisine:1})

19. Display the same fields for restaurants not in those four boroughs.

     db.restaurants.find({borough:{$nin:["Staten Island","Queens","Bronx","Brooklyn"]}},{restaurant_id:1,name:1,borough:1,cuisine:1})

20. Find restaurant_id, name, borough, and cuisine for restaurants with a score not more than 10.

     db.restaurants.find({"grades.score":{$lte:10}},{restaurant_id:1,name:1,borough:1,cuisine:1})
    
21. Find restaurant_id, name, borough, and cuisine for restaurants that:
     Serve neither American nor Chinese cuisine, OR
     Have a name starting with 'Wil'
   
     db.restaurants.find({$or:[{cuisine:{$nin:["American","Chinese"]}},{name:/^Wil/}]},{restaurant_id:1,name:1,borough:1,cuisine:1})

Q22. Find restaurant_id, name, and grades for restaurants that received:
      Grade "A"
      Score 11
      On 2014-08-11T00:00:00Z

     db.restaurants.find({grades:{$elemMatch:{grade:"A",score:11,date:ISODate('2014-08-11T00:00:00Z')}}},{restaurant_id:1,name:1,grades:1})

Q23. Find restaurant_id, name, and grades where the 2nd element of the grades array has:
      Grade "A"
      Score 9
      Date 2014-08-11T00:00:00Z

     db.restaurants.find({"grades.1.grade":"A","grades.1.score":11,"grades.1.date":ISODate('2014-08-11T00:00:00Z')},{restaurant_id:1,name:1,grades:1})

Q24. Find restaurant_id, name, address, and geo-location for restaurants where the 2nd element of the coord array is more than 42 and up to 52.

     db.restaurants.find({"address.coord.1":{$gt:42,$lte:52}},{restaurant_id:1,name:1,address:1})

25. List all restaurants sorted by name ascending.

     db.restaurants.find().sort({name:1})

Q26. List all restaurants sorted by name descending.

     db.restaurants.find().sort({name:-1})

Q27. Sort by cuisine ascending, and within the same cuisine sort by borough descending.

     db.restaurants.find().sort({cuisine:1,borough:-1})

28. Check whether all address documents contain a street field.

     db.restaurants.find({"address.street":{$exists:true}})


Q29. Select all documents where the coord field value is of type Double.
  
     db.restaurants.find({"address.coord.0":{$type:"double"},"address.coord.1":{$type:"double"}})

30. Find restaurant_id, name, and grades for restaurants where the score is divisible by 7 (remainder = 0).

     db.restaurants.find({"grades.score":{$mod:[7,0]}},{restaurant_id:1,name:1,grades:1})
     db.restaurants.find({grades:{$elemMatch:{score:{$mod:[7,0]}}}},{restaurant_id:1,name:1,grades:1})

     db.restaurants.find({name:/mon/},{name:1,borough:1,"address.coord":1,cuisine:1})
